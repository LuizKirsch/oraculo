# ====================================
# COMANDOS PARA GERAR O BANCO VETORIAL
# ====================================

# 1. Parar a aplicação (se estiver rodando)
docker-compose down

# 2. Limpar o banco vetorial atual
Remove-Item -Recurse -Force "db_oraculo\*" -ErrorAction SilentlyContinue

# 3. Construir a imagem Docker
docker-compose build

# 4. Executar a ingestão dos documentos
docker-compose run --rm oraculo-app python -m app.ingestao

# 5. Iniciar a aplicação
docker-compose up -d

# 6. Verificar se está funcionando
docker-compose ps
docker-compose logs --tail=10

# ====================================
# NOTAS:
# - Execute estes comandos sempre que:
#   * Adicionar novos documentos na pasta oraculo_docs
#   * Modificar a função carregar_documentos() no oraculo.py
#   * Precisar recriar o banco vetorial do zero
