# =======================================
# COMANDOS PARA ATUALIZAR CÓDIGO/DOCKER
# =======================================

# 1. Parar a aplicação
docker-compose down

# 2. Reconstruir a imagem Docker com as mudanças
docker-compose build

# 3. Iniciar a aplicação
docker-compose up -d

# 4. Verificar se está funcionando
docker-compose ps

# 5. (Opcional) Ver logs em tempo real
docker-compose logs -f

# =======================================
# COMANDOS ÚTEIS EXTRAS:
# =======================================

# Ver logs das últimas 20 linhas:
docker-compose logs --tail=20

# Parar e remover tudo (limpeza completa):
docker-compose down --volumes --remove-orphans

# Executar apenas a ingestão (sem rebuild):
docker-compose run --rm oraculo-app python -m app.ingestao

# Entrar no container para debug:
docker-compose exec oraculo-app bash
